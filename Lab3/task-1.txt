(defrule rule-1-father
	(declare (salience 90))
	(parent-of (parent-type father) (parent ?p) (child ?c))
	=>
	(printout t "The father of " ?c " is " ?p "." crlf))
 
(defrule rule-2-mother
	(declare (salience 80))
	(parent-of (parent-type mother) (parent ?p) (child ?c))
	=>
	(printout t "The mother of " ?c " is " ?p "." crlf))
 
(defrule rule-3-parents
	(declare (salience 70))
	(parents-of (child ?c) (parents ?p1 ?p2))
	=>
	(printout t "The parents of " ?c " are " ?p1 " and " ?p2 "." crlf))
 
(defrule rule-4-tom-father
	(declare (salience 60))
	(has-role (person Tom) (role father))
	=>
	(printout t "Tom is a father." crlf))
