;; Опис зірки
(deftemplate star
	(slot name)
	(slot class (allowed-values A B F G K M O))
	(slot magnitude (type INTEGER))
	(slot distance (type FLOAT))
)

;; Вхідні дані - заданий клас і величина
(deftemplate query-class
	(slot class (allowed-values A B F G K M O))
)

(deftemplate query-magnitude
	(slot magnitude (type INTEGER))
)

;; Вхідні факти - два факти, що задають клас і величину для пошуку
(deffacts input-query
	(query-class (class A))
	(query-magnitude (magnitude 1))
)

;; --- Блок 1: усі зірки заданого спектрального класу ---
(defrule header-class
	(declare (salience 35))
	(query-class (class ?cl))
	=>
	(printout t "=== Зорі спектрального класу " ?cl " ===" crlf))

(defrule show-by-class
	(declare (salience 30))
	(query-class (class ?cl))
	(star (name ?n) (class ?cl))
	=>
	(printout t ?n " - клас " ?cl crlf))

;; --- Блок 2: усі зірки заданої величини ---
(defrule header-magnitude
	(declare (salience 25))
	(query-magnitude (magnitude ?m))
	=>
	(printout t "=== Зорі величини " ?m " ===" crlf))

(defrule show-by-magnitude
	(declare (salience 20))
	(query-magnitude (magnitude ?m))
	(star (name ?n) (magnitude ?m))
	=>
	(printout t ?n " - величина " ?m crlf))

;; --- Блок 3: зірки, що відповідають обом умовам + відстань ---
(defrule header-both
	(declare (salience 15))
	(query-class (class ?cl))
	(query-magnitude (magnitude ?m))
	=>
	(printout t "=== Зорі класу " ?cl " та величини " ?m " (з відстанню) ===" crlf))

(defrule show-by-both
	(declare (salience 10))
	(query-class (class ?cl))
	(query-magnitude (magnitude ?m))
	(star (name ?n) (class ?cl) (magnitude ?m) (distance ?d))
	=>
	(printout t ?n " - клас " ?cl ", величина " ?m ", відстань " ?d " св. років" crlf))
